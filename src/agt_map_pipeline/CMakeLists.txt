cmake_minimum_required(VERSION 3.8)
project(agt_map_pipeline)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
set(AGT_MAP_PIPELINE_CLI_DIR "${CMAKE_CURRENT_BINARY_DIR}/cli")
file(MAKE_DIRECTORY "${AGT_MAP_PIPELINE_CLI_DIR}")
configure_file(scripts/agt_map_launcher.py "${AGT_MAP_PIPELINE_CLI_DIR}/agt_map_launcher.py" COPYONLY)
file(COPY "${AGT_MAP_PIPELINE_CLI_DIR}/agt_map_launcher.py" DESTINATION "${AGT_MAP_PIPELINE_CLI_DIR}" FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
install(PROGRAMS "${AGT_MAP_PIPELINE_CLI_DIR}/agt_map_launcher.py" DESTINATION bin RENAME agt-map)
install(PROGRAMS "${AGT_MAP_PIPELINE_CLI_DIR}/agt_map_launcher.py" DESTINATION lib/${PROJECT_NAME} RENAME agt-map)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_project_contract test/test_project_contract.py)
  ament_add_pytest_test(test_presets test/test_presets.py)
  ament_add_pytest_test(test_layer_io test/test_layer_io.py)
  ament_add_pytest_test(test_prepare test/test_prepare.py)
  ament_add_pytest_test(test_status_cli test/test_status_cli.py)
  ament_add_pytest_test(test_prepare_e2e test/test_prepare_e2e.py)
  ament_add_pytest_test(test_canonical_frame test/test_canonical_frame.py)
  ament_add_pytest_test(test_frame_verification test/test_frame_verification.py)
endif()
ament_package()
